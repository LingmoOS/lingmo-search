cmake_minimum_required(VERSION 3.22)
project(lingmo-search VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CTest)
find_package(LingmoSDK REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ICU REQUIRED IMPORTED_TARGET icu-uc)
add_library(search-core STATIC src/search.cpp src/expression.cpp src/settings_catalog.cpp)
target_include_directories(search-core PUBLIC src)
target_link_libraries(search-core PUBLIC Lingmo::SDKBase PRIVATE PkgConfig::ICU)
add_library(search-ui STATIC src/searchwindow.cpp)
target_link_libraries(search-ui PUBLIC search-core Lingmo::SDK)
add_executable(lingmo-search src/main.cpp)
target_link_libraries(lingmo-search PRIVATE search-ui)
install(TARGETS lingmo-search RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
configure_file(data/org.lingmo.Search.service.in org.lingmo.Search.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.lingmo.Search.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
lingmo_add_translations(lingmo-search SOURCES src DESKTOP_FILES lingmo-search.desktop)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/lingmo-search.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
